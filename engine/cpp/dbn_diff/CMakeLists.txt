# dbn_dump — differential-oracle DBN dumper on top of the OFFICIAL databento-cpp.
#
# This CMake project is deliberately standalone (it is NOT part of the
# /workspace/engine/cpp tree build) so that the vendored official library can be
# added verbatim via add_subdirectory without perturbing the qr_* targets.
#
# Configure/build out-of-tree, e.g.:
#   cmake -S /workspace/engine/cpp/dbn_diff \
#         -B /workspace/artifacts/cache/cpp/dbn_diff/build -DCMAKE_BUILD_TYPE=Release
#   cmake --build /workspace/artifacts/cache/cpp/dbn_diff/build -j
#
# Build outputs MUST live under /workspace/artifacts/cache/ (D-018); never commit them.

cmake_minimum_required(VERSION 3.24)
project(dbn_diff LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(DBN_DIFF_VENDOR_DIR "/workspace/vendor/databento-cpp"
    CACHE PATH "Vendored official databento-cpp checkout (read-only)")

# Full official library (needs OpenSSL/httplib/nlohmann_json/date) unless the caller
# asks for the trimmed decode-only route.
option(DBN_DIFF_DECODE_ONLY
       "Compile only the DBN+zstd decode translation units instead of the full library"
       OFF)

if(NOT DBN_DIFF_DECODE_ONLY)
  # The vendored project defaults IS_MAIN_PROJECT=OFF here, so warnings-as-errors,
  # examples, tests and verbose output all stay off.
  add_subdirectory("${DBN_DIFF_VENDOR_DIR}" "${CMAKE_BINARY_DIR}/databento-cpp" EXCLUDE_FROM_ALL)
  add_executable(dbn_dump "${CMAKE_CURRENT_LIST_DIR}/dbn_dump.cpp")
  target_link_libraries(dbn_dump PRIVATE databento::databento)
else()
  # Decode-only fallback: just the translation units reachable from DbnFileStore.
  # It drops the historical/live/tcp/http client TUs, but NOT the dependencies:
  # databento/exceptions.hpp is pulled in by record.hpp, and it includes both
  # <httplib.h> and <nlohmann/json.hpp> *and* force-defines CPPHTTPLIB_OPENSSL_SUPPORT
  # before the httplib include. So every decode TU still needs the httplib, json and
  # date headers and still links OpenSSL. Point DBN_DIFF_DATE_INCLUDE_DIR /
  # DBN_DIFF_JSON_INCLUDE_DIR at a directory holding date/date.h, nlohmann/json.hpp
  # and httplib.h (build.sh --decode-only fetches all three).
  find_package(Threads REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(zstd QUIET)
  if(NOT TARGET zstd::libzstd AND NOT TARGET zstd::libzstd_shared
     AND NOT TARGET zstd::libzstd_static)
    find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
  endif()

  set(_dbn_srcs
    src/datetime.cpp
    src/dbn.cpp
    src/dbn_decoder.cpp
    src/dbn_store.cpp
    src/detail/buffer.cpp
    src/detail/json_helpers.cpp
    src/detail/zstd_stream.cpp
    src/enums.cpp
    src/exceptions.cpp
    src/file_stream.cpp
    src/flag_set.cpp
    src/log.cpp
    src/metadata.cpp
    src/pretty.cpp
    src/publishers.cpp
    src/record.cpp
    src/symbol_map.cpp
    src/v1.cpp
    src/v2.cpp
  )
  list(TRANSFORM _dbn_srcs PREPEND "${DBN_DIFF_VENDOR_DIR}/")

  # version.hpp / system.hpp are configure_file-generated by the vendored project.
  set(DATABENTO_VERSION "0.64.0")
  set(PROJECT_VERSION_MAJOR 0)
  set(PROJECT_VERSION_MINOR 64)
  set(PROJECT_VERSION_PATCH 0)
  configure_file("${DBN_DIFF_VENDOR_DIR}/cmake/version.hpp.in"
                 "${CMAKE_BINARY_DIR}/gen/databento/version.hpp" @ONLY)
  configure_file("${DBN_DIFF_VENDOR_DIR}/cmake/system.hpp.in"
                 "${CMAKE_BINARY_DIR}/gen/databento/system.hpp" @ONLY)

  add_executable(dbn_dump "${CMAKE_CURRENT_LIST_DIR}/dbn_dump.cpp" ${_dbn_srcs})
  target_include_directories(dbn_dump PRIVATE
    "${DBN_DIFF_VENDOR_DIR}/include"
    "${DBN_DIFF_VENDOR_DIR}/src"
    "${CMAKE_BINARY_DIR}/gen"
    "${DBN_DIFF_DATE_INCLUDE_DIR}"
    "${DBN_DIFF_JSON_INCLUDE_DIR}"
  )
  if(TARGET zstd::libzstd)
    target_link_libraries(dbn_dump PRIVATE zstd::libzstd)
  elseif(TARGET zstd::libzstd_shared)
    target_link_libraries(dbn_dump PRIVATE zstd::libzstd_shared)
  elseif(TARGET zstd::libzstd_static)
    target_link_libraries(dbn_dump PRIVATE zstd::libzstd_static)
  else()
    target_link_libraries(dbn_dump PRIVATE "${ZSTD_LIBRARY}")
  endif()
  target_link_libraries(dbn_dump PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
endif()

set_target_properties(dbn_dump PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# ---------------------------------------------------------------------------
# qr_dbn_dump — the SAME dump produced by OUR OWN hand-written decoder
# (/workspace/engine/cpp/qr_dbn). Third leg of the three-way differential.
#
# qr_dbn is compiled here from its own sources rather than via the qr_substrate
# tree build, so that this standalone project stays standalone and so that
# nothing under /workspace/engine/cpp/qr_dbn or qr_core is modified. The
# translation units are exactly the ones qr_dbn's own CMakeLists lists, plus the
# single qr_core TU (refusal.cpp) they depend on.
#
# Build:
#   cmake --build /workspace/artifacts/cache/cpp/dbn_diff/build --target qr_dbn_dump -j
# ---------------------------------------------------------------------------
set(QR_ENGINE_CPP_DIR "/workspace/engine/cpp" CACHE PATH "qr_* substrate sources (read-only)")

find_package(Threads REQUIRED)
find_path(QR_DBN_ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(QR_DBN_ZSTD_LIBRARY NAMES zstd REQUIRED)

add_executable(qr_dbn_dump
  "${CMAKE_CURRENT_LIST_DIR}/qr_dbn_dump.cpp"
  "${QR_ENGINE_CPP_DIR}/qr_dbn/src/dbn.cpp"
  "${QR_ENGINE_CPP_DIR}/qr_dbn/src/zstd_stream.cpp"
  "${QR_ENGINE_CPP_DIR}/qr_core/src/refusal.cpp"
)
target_include_directories(qr_dbn_dump PRIVATE
  "${QR_ENGINE_CPP_DIR}/qr_dbn/include"
  "${QR_ENGINE_CPP_DIR}/qr_core/include"
  "${QR_DBN_ZSTD_INCLUDE_DIR}"
)
target_link_libraries(qr_dbn_dump PRIVATE "${QR_DBN_ZSTD_LIBRARY}" Threads::Threads)
# The substrate builds these TUs at C++20 with the boundary warning wall; keep
# both so the code under test is compiled exactly as it is in its home tree.
set_target_properties(qr_dbn_dump PROPERTIES
  CXX_STANDARD 20
  CXX_STANDARD_REQUIRED ON
  CXX_EXTENSIONS OFF
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_options(qr_dbn_dump PRIVATE
  -Wall -Wextra -Wpedantic -Werror -ffp-contract=off
  -Werror=conversion -Werror=sign-conversion -Wconversion -Wsign-conversion)
